using System.ComponentModel.Design;

const int maxmertebe = 10;
const int minmertebe = 1;
int cariLiftinDayandigiMertebe = 1;
while (true)
{
    Console.WriteLine($"Hal- hazirda {cariLiftinDayandigiMertebe} ci mertebedesiz");
    Console.Write("Gedeceyiniz mertebeni daxil edin : ");
    int mertebe = int.Parse(Console.ReadLine());
    if (mertebe == 0)
    {
        break;
    }
    if (mertebe < minmertebe || mertebe > maxmertebe)
    {
        Console.WriteLine("Bele mertebe yoxdu");
        continue;
    }
    if (mertebe == cariLiftinDayandigiMertebe)
    {
        Console.WriteLine("Eyni mertebedesiniz");

    }
    else if (mertebe > cariLiftinDayandigiMertebe)
    {
        for (int floor = cariLiftinDayandigiMertebe + 1; floor <= mertebe;floor++)
        {
            Console.WriteLine($"{floor}- ci mertebe");
        }
        cariLiftinDayandigiMertebe = mertebe;
        Console.WriteLine("Catdiniz");
    }
    else if (mertebe < cariLiftinDayandigiMertebe)
    {
        for (int floor = cariLiftinDayandigiMertebe - 1; floor >= mertebe; floor--)
        {
            Console.WriteLine($"{floor}- ci mertebe");
        }
        cariLiftinDayandigiMertebe = mertebe;
        Console.WriteLine("Catdiniz");
    }
}

Console.Write("Şifrə daxil edin: ");
string sifre = Console.ReadLine();
int boyukherfsayi = 0;
int kicikherfsayi = 0;
int reqemsayi = 0;
foreach (char simvol in sifre)
{
if (char.IsUpper(simvol))
{
boyukherfsayi++;
}
else if (char.IsLower(simvol))
{
kicikherfsayi++;
}
else if (char.IsDigit(simvol))
{
reqemsayi++;
}
}
Console.WriteLine($"Böyük hərf sayı: {boyukherfsayi}");
Console.WriteLine($"Kiçik hərf sayı: {kicikherfsayi}");
Console.WriteLine($"Rəqəm sayı: {reqemsayi}");
