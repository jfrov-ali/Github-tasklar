Task 2
Console.Write("Adi daxil edin : ");
string ad = Console.ReadLine();
Dictionary<string, int> telebeler = new()
{
    ["Aysel"] = 23,
    ["Maqa"] = 25,
    ["Kamal"] = 27,
    ["Muhammed"] = 28
};
bool tapildimi = false;
foreach(KeyValuePair<string,int> telebe in telebeler)
{
    if (telebe.Key == ad)
    {
        Console.WriteLine($"Ad tapildi, daxil etdiyiniz ad {ad} adli telebenin bali: {telebe.Value}");
        tapildimi = true;
        break;
    }
}
if ( ! tapildimi)
{
    Console.WriteLine("Ad tapilmadi");
}
